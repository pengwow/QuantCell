# dual_ma_btc_4h.py — 双均线趋势策略（示例 · 金叉买入 / 死叉卖出）from dataclasses import dataclassfrom collections import dequefrom axon_quant import Action, ActionType@dataclassclass DualMAParams:    fast: int = 20          # 快线窗口（bar 数）    slow: int = 60          # 慢线窗口    target: float = 0.10    # 目标仓位（权益占比）class DualMAStrategy:    """金叉买入、死叉卖出；其余 bar 维持持仓。"""    def __init__(self, params: DualMAParams | None = None):        self.p = params or DualMAParams()        self.closes = deque(maxlen=self.p.slow)        self.pos: float = 0.0    def on_bar(self, bar: dict) -> Action:        self.closes.append(bar["close"])        if len(self.closes) < self.p.slow:            return Action(ActionType.Hold, 0.0, self.pos, "dual_ma", 0)        ma_fast = sum(list(self.closes)[-self.p.fast:]) / self.p.fast        ma_slow = sum(self.closes) / self.p.slow        if ma_fast > ma_slow:   # 金叉 → 买入            self.pos = self.p.target            return Action(ActionType.Buy, 0.8, self.p.target, "dual_ma", 0)        elif ma_fast < ma_slow: # 死叉 → 清仓            self.pos = 0.0            return Action(ActionType.Sell, 0.8, 0.0, "dual_ma", 0)        return Action(ActionType.Hold, 0.0, self.pos, "dual_ma", 0)
